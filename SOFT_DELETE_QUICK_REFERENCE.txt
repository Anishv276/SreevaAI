┌────────────────────────────────────────────────────────────────────────────┐
│ THE FIX PATTERN                                                            │
├────────────────────────────────────────────────────────────────────────────┤
│                                                                            │
│ SIMPLE COUNT:                                                              │
│   ❌ SELECT COUNT(*) FROM "Drive"                                         │
│   ✅ SELECT COUNT(*) FROM "Drive" WHERE "isDeleted" = false               │
│                                                                            │
│ MULTIPLE CONDITIONS:                                                       │
│   ❌ SELECT COUNT(*) FROM "Drive" WHERE "orgId" = '123'                   │
│   ✅ SELECT COUNT(*) FROM "Drive"                                         │
│      WHERE "orgId" = '123' AND "isDeleted" = false                        │
│                                                                            │
│ JOIN QUERIES:                                                              │
│   ❌ FROM "Organization" o                                                │
│      JOIN "Drive" d ON d."orgId" = o.id                                   │
│                                                                            │
│   ✅ FROM "Organization" o                                                │
│      JOIN "Drive" d ON d."orgId" = o.id                                   │
│      WHERE o."isDeleted" = false AND d."isDeleted" = false               │
│                                                                            │
│ SUBQUERIES:                                                                │
│   ❌ (SELECT COUNT(*) FROM "User" u WHERE u."orgId" = o.id)              │
│                                                                            │
│   ✅ (SELECT COUNT(*) FROM "User" u                                       │
│       WHERE u."orgId" = o.id AND u."isDeleted" = false)                  │
│                                                                            │

┌────────────────────────────────────────────────────────────────────────────┐
│ COMMON MISTAKES TO AVOID                                                   │
├────────────────────────────────────────────────────────────────────────────┤
│                                                                            │
│ ❌ Using OR condition: WHERE "isDeleted" = false OR x = y                 │
│    (This defeats soft-delete if x = y matches deleted records)            │
│                                                                            │
│ ✅ Using AND: WHERE x = y AND "isDeleted" = false                         │
│                                                                            │
│ ❌ Forgetting filter in JOINs: Only filtering main table                  │
│    (Deleted related records still appear)                                 │
│                                                                            │
│ ✅ Filter ALL tables: WHERE o."isDeleted" = false                         │
│                       AND d."isDeleted" = false                           │
│                                                                            │
│ ❌ Filtering only in WHERE, not in subqueries                             │
│                                                                            │
│ ✅ Filter everywhere: Main query + subqueries + JOINs                    │
│                                                                            │
