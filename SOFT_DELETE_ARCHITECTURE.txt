AFFECTED QUERY PATTERNS
=======================

Pattern 1: Direct COUNT with no WHERE
─────────────────────────────────────
❌ (SELECT COUNT(*)::INT FROM "Drive")
✅ (SELECT COUNT(*)::INT FROM "Drive" WHERE "isDeleted" = false)

Pattern 2: JOIN with multiple tables
─────────────────────────────────────
❌ FROM "Organization" o
   JOIN "Drive" d ON d."orgId" = o.id
   JOIN "Application" a ON a."driveId" = d.id
   JOIN "Interview" i ON i."applicationId" = a.id

✅ FROM "Organization" o
   JOIN "Drive" d ON d."orgId" = o.id
   JOIN "Application" a ON a."driveId" = d.id
   JOIN "Interview" i ON i."applicationId" = a.id
   WHERE o."isDeleted" = false
     AND d."isDeleted" = false
     AND a."isDeleted" = false
     AND i."isDeleted" = false

Pattern 3: Subqueries with counts
──────────────────────────────────
❌ (SELECT COUNT(*)::INT FROM "User" u WHERE u."orgId" = o.id)

✅ (SELECT COUNT(*)::INT FROM "User" u WHERE u."orgId" = o.id AND u."isDeleted" = false)

Pattern 4: UNION ALL with multiple source tables
─────────────────────────────────────────────────
❌ SELECT 'ORG'::text AS type, o.name FROM "Organization" o
   UNION ALL
   SELECT 'DRIVE'::text AS type, d.title FROM "Drive" d

✅ SELECT 'ORG'::text AS type, o.name FROM "Organization" o WHERE o."isDeleted" = false
   UNION ALL
   SELECT 'DRIVE'::text AS type, d.title FROM "Drive" d WHERE d."isDeleted" = false
